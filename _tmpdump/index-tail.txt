ijnseNaam || '', 'nl');
            });
            const visible = sorted.filter(slMatches);

            slVisibleInfoEl.textContent = visible.length === sorted.length ? '' :
                (visible.length + ' van ' + sorted.length + ' plant(en) getoond');

            const genusCounts = {};
            visible.forEach(function (p) { const g = genusOf(p); genusCounts[g] = (genusCounts[g] || 0) + 1; });

            let currentGenus = null;
            const rows = visible.map(function (plant) {
                const name = esc(plant.latijnseNaam);
                const genus = genusOf(plant);
                let html = '';
                if (genus !== currentGenus) {
                    currentGenus = genus;
                    html += '<tr class="sl-genus"><td colspan="19">' + esc(genus) +
                        ' <span class="sl-genus-count">(' + (genusCounts[genus] || 0) + ')</span></td></tr>';
                }
                // standplaats in de juiste kolom (Z / HS / S)
                const tokens = slTokens(plant);
                const spCols = ['Z', 'HS', 'S'].map(function (t) {
                    const ja = tokens.indexOf(t) !== -1;
                    return '<td class="sl-sp' + (ja ? ' ja' : '') + '">' +
                        (ja ? t : (tokens.length ? '' : '&middot;')) + '</td>';
                }).join('');
                // hoogtebalk 0-2 m met streepjes + kleine tekst
                const van = Number(plant.hoogteVan), tot = Number(plant.hoogteTot);
                const topCm = (tot > 0 ? tot : 0) || (van > 0 ? van : 0);
                const pct = topCm > 0 ? Math.min(topCm, 200) / 2 : 0;
                const hTxt = (van > 0 && tot > van) ? (van + '\u2013' + tot + ' cm')
                    : (topCm > 0 ? (topCm + ' cm') : 'onbekend');
                // 12 maandkolommen met de beginletter als de plant bloeit
                const bloei = PlantPicker.bloeiMaanden(plant);
                const maandCols = PlantPicker.MONTHS.map(function (m, i) {
                    const bloeit = bloei.indexOf(i) !== -1;
                    return '<td class="sl-m' + (bloeit ? ' bloei' : '') + '"' +
                        ' title="' + esc(m) + (bloeit ? ' \u2013 bloeit' : '') + '">' +
                        (bloeit ? esc(m.charAt(0)) : '') + '</td>';
                }).join('');
                return html +
                    '<tr>' +
                    '<td class="sl-plant"><strong>' + name + '</strong><br><em>' + esc(plant.nlNaam || '') + '</em></td>' +
                    '<td class="sl-kleur"><span class="color-indicator" style="background-color: ' + getColorHex(plant.kleur) + '"></span> ' + esc(plant.kleur || '-') + '</td>' +
                    spCols +
                    '<td class="sl-hoogte" title="' + esc(hTxt) + '">' +
                    '<div class="sl-hoogte-bar"><div class="sl-hoogte-fill" style="width: ' + pct + '%"></div></div>' +
                    '<span class="sl-hoogte-txt">' + esc(hTxt) + '</span></td>' +
                    maandCols +
                    '<td class="sl-del"><button data-remove="' + name + '">\u2715</button></td>' +
                    '</tr>';
            }).join('');

            shortlistEl.innerHTML = rows ||
                '<tr><td colspan="19" class="sl-empty">' +
                (shortlist.length
                    ? 'Geen planten voldoen aan de filters.'
                    : 'Nog geen planten op de shortlist &ndash; vink planten aan in het Planten-tabblad.') +
                '</td></tr>';
            updateAddPlantSelect();
        }

        shortlistEl.addEventListener('click', function (e) {
            if (e.target && e.target.tagName === 'BUTTON' && e.target.hasAttribute('data-remove')) {
                removeFromShortlist(e.target.getAttribute('data-remove'));
            }
        });

        function removeFromShortlist(latijnseNaam) {
            shortlist = shortlist.filter(p => p.latijnseNaam !== latijnseNaam);
            saveState();
            plantPicker.refresh();
            renderShortlist(); }

        function clearShortlist() {
            if (confirm('Shortlist leegmaken?')) {
                shortlist = [];
                saveState();
                plantPicker.refresh();
                renderShortlist();
            }
        }

        function addNewBorder() {
            const borderName = prompt('Naam voor nieuwe border:');
            if (!borderName) return;
            const newBorder = { id: Date.now().toString(), name: borderName, plants: [], shape: null };
            borders.push(newBorder);
            activeBorderId = newBorder.id;
            saveState();
            renderBorderList();
            showActiveBorder(newBorder);
        }

        function showActiveBorder(borderOrId) {
            const border = typeof borderOrId === 'object' ? borderOrId : borders.find(b => b.id === borderOrId);
            if (!border) return;
            activeBorderId = border.id;
            activeBorderNameEl.textContent = 'Border: ' + border.name;
            activeBorderSectionEl.classList.remove('hidden');
            renderBorderPlantTable();
            updateAddPlantSelect();
        }

        function renderBorderList() {
            borderListEl.innerHTML = borders.map(border =>
                '<div class="border-item ' + (activeBorderId === border.id ? 'active' : '') + '" data-border="' + esc(border.id) + '">' +
                '<span><strong>' + esc(border.name) + '</strong> (' + border.plants.length + ' planten' + (border.shape && border.shape.length >= 3 ? ', vorm getekend' : '') + ')</span>' +
                '<span>' +
                '<button data-editborder="' + esc(border.id) + '">&#9998; Bewerken</button>' +
                '<button data-deleteborder="' + esc(border.id) + '">\u2715</button>' +
                '</span>' +
                '</div>'
            ).join('');
        }

        borderListEl.addEventListener('click', function (e) {
            if (e.target && e.target.tagName === 'BUTTON' && e.target.hasAttribute('data-deleteborder')) {
                e.stopPropagation();
                deleteBorder(e.target.getAttribute('data-deleteborder'));
                return;
            }
            if (e.target && e.target.tagName === 'BUTTON' && e.target.hasAttribute('data-editborder')) {
                e.stopPropagation();
                window.location.href = 'border.html?id=' + encodeURIComponent(e.target.getAttribute('data-editborder'));
                return;
            }
            const item = e.target.closest ? e.target.closest('.border-item') : null;
            if (item && item.hasAttribute('data-border')) {
                showActiveBorder(item.getAttribute('data-border'));
            }
        });

        function renderBorderPlantTable() {
            const border = borders.find(b => b.id === activeBorderId);
            if (!border) return;
            borderPlantTableBodyEl.innerHTML = border.plants.map(plant => {
                const colorHex = getColorHex(plant.kleur);
                const name = esc(plant.latijnseNaam);
                return '<tr>' +
                    '<td><strong>' + name + '</strong><br><em>' + esc(plant.nlNaam) + '</em></td>' +
                    '<td><span class="color-indicator" style="background-color: ' + colorHex + '"></span> ' + esc(plant.kleur) + '</td>' +
                    '<td>' + esc(plant.bloeiVan || '-') + '</td>' +
                    '<td>' + esc(plant.bloeiTot || '-') + '</td>' +
                    '<td>' + (plant.hoogteVan || '-') + 'cm</td>' +
                    '<td>' + (plant.hoogteTot > plant.hoogteVan ? plant.hoogteTot + 'cm' : '') + '</td>' +
                    '<td>' + esc(plant.standplaats || '-') + '</td>' +
                    '<td><button class="btn btn-danger" data-removefromborder="' + name + '">\u2715</button></td>' +
                    '</tr>'; }).join('');
        }

        borderPlantTableBodyEl.addEventListener('click', function (e) {
            if (e.target && e.target.tagName === 'BUTTON' && e.target.hasAttribute('data-removefromborder')) {
                removePlantFromBorder(e.target.getAttribute('data-removefromborder'));
            }
        });

        function updateAddPlantSelect() {
            const border = borders.find(b => b.id === activeBorderId);
            const availablePlants = shortlist.filter(p =>
                !border || !border.plants.some(bp => bp.latijnseNaam === p.latijnseNaam)
            );
            addPlantSelectEl.innerHTML = '<option value="">Selecteer een plant...</option>' +
                availablePlants.map(plant =>
                    '<option value="' + esc(plant.latijnseNaam) + '">' + esc(plant.latijnseNaam) + ' - ' + esc(plant.nlNaam) + '</option>'
                ).join(''); }

        function addPlantToBorder() {
            const selected = addPlantSelectEl.value;
            if (!selected) return;
            const plant = shortlist.find(p => p.latijnseNaam === selected);
            if (!plant) return;
            const border = borders.find(b => b.id === activeBorderId);
            if (!border) return;
            const copy = JSON.parse(JSON.stringify(plant));
            copy.pos = null;
            border.plants.push(copy);
            saveState();
            renderBorderPlantTable();
            updateAddPlantSelect();
            addPlantSelectEl.value = ''; }

        function removePlantFromBorder(latijnseNaam) {
            const border = borders.find(b => b.id === activeBorderId);
            if (!border) return;
            border.plants = border.plants.filter(p => p.latijnseNaam !== latijnseNaam);
            saveState();
            renderBorderPlantTable();
            updateAddPlantSelect(); }

        function clearBorder() {
            const border = borders.find(b => b.id === activeBorderId);
            if (!border) return;
            if (confirm('Border leegmaken?')) {
                border.plants = [];
                saveState();
                renderBorderPlantTable();
                updateAddPlantSelect();
            }
        }

        function deleteBorder(borderId) {
            if (confirm('Border verwijderen?')) {
                borders = borders.filter(b => b.id !== borderId);
                if (activeBorderId === borderId) {
                    activeBorderId = null;
                    activeBorderSectionEl.classList.add('hidden');
                }
                saveState();
                renderBorderList();
            }
        }

        function closeBorder() {
            activeBorderId = null;
            activeBorderSectionEl.classList.add('hidden');
        }

        function saveState() {
            localStorage.setItem('shortlist', JSON.stringify(shortlist));
            localStorage.setItem('borders', JSON.stringify(borders));
        }

        // Tuinontwerp: lijst met tuinobjecten (gardenObjects)
        const GARDEN_OBJECT_TYPES = {
            terras: '#c8a06a', pad: '#a08a6a', gazon: '#7cb342', border: '#e57373', haag: '#388e3c',
            moestuin: '#8d6e63', boom: '#2e7d32', vijver: '#4fc3f7', zithoek: '#ffb74d', zandbak: '#fdd835',
            speeltoestel: '#f06292', vuurplaats: '#ff7043', plantebak: '#9575cd', techniek: '#90a4ae', misc: '#bdbdbd'
        };

        function readGardenObjects() {
            try { return JSON.parse(localStorage.getItem('gardenObjects')) || []; } catch (e) { return []; }
        }

        function renderGardenObjectList() {
            const listEl = document.getElementById('gardenObjectList');
            const objects = readGardenObjects();
            if (!objects.length) {
                listEl.innerHTML = '<p class="settings-note">Nog geen tuinobjecten. Importeer een ontwerp via Instellingen (&#9881;) &rarr; Importeer JSON, of teken borders op de kaart.</p>';
                return;
            }
            listEl.innerHTML = objects.map(obj =>
                '<div class="border-item">' +
                '<span><span class="color-indicator" style="background-color: ' + (obj.color || GARDEN_OBJECT_TYPES[obj.type] || '#cccccc') + '"></span> <strong>' + esc(obj.name || obj.type) + '</strong> <em style="color: var(--light-text);">(' + esc(obj.type) + (obj.materiaal ? ' &middot; ' + esc(obj.materiaal) : '') + ')</em></span>' +
                '<span><button data-deleteobject="' + esc(obj.id) + '">✕</button></span>' +
                '</div>'
            ).join('');
        }

        document.getElementById('gardenObjectList').addEventListener('click', function (e) {
            if (e.target && e.target.tagName === 'BUTTON' && e.target.hasAttribute('data-deleteobject')) {
                const id = e.target.getAttribute('data-deleteobject');
                if (confirm('Tuinobject verwijderen?')) {
                    localStorage.setItem('gardenObjects', JSON.stringify(readGardenObjects().filter(o => o.id !== id)));
                    renderGardenObjectList();
                    if (typeof window.renderMap === 'function') window.renderMap();
                }
            }
        });

        // Tabs (Planten / Borders / Shortlist)
        const tabsEl = document.getElementById('mainTabs');
        function showTab(name) {
            tabsEl.querySelectorAll('button').forEach(function (b) {
                b.classList.toggle('active', b.getAttribute('data-tab') === name);
            });
            document.querySelectorAll('[data-tabpane]').forEach(function (p) {
                p.classList.toggle('active', p.getAttribute('data-tabpane') === name);
            });
            try { localStorage.setItem('activeTab', name); } catch (e) { }
            if (name === 'ontwerp' && window.gardenMap) {
                setTimeout(function () { gardenMap.invalidateSize(); }, 50);
                setTimeout(function () { gardenMap.invalidateSize(); }, 400);
            }
        }
        tabsEl.addEventListener('click', function (e) {
            const btn = e.target.closest ? e.target.closest('button[data-tab]') : null;
            if (btn) showTab(btn.getAttribute('data-tab'));
        });
        let activeTab = 'planten';
        try { activeTab = localStorage.getItem('activeTab') || 'planten'; } catch (e) { }
        showTab(activeTab);

        // Event listeners
        document.getElementById('clearShortlistBtn').addEventListener('click', clearShortlist);
        document.getElementById('addBorderBtn').addEventListener('click', addNewBorder);
        document.getElementById('addPlantToBorderBtn').addEventListener('click', addPlantToBorder);
        document.getElementById('closeBorderBtn').addEventListener('click', closeBorder);
        document.getElementById('clearBorderBtn').addEventListener('click', clearBorder);

        // Init: render alles bij het laden van de pagina
        plantPicker.refresh();
        renderShortlist();
        renderBorderList();
        renderGardenObjectList();
    </script>
</body>
